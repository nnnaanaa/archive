; --- 設定ファイルのパス ---
INI_FILE = 'setting.ini'

; --- ファイルを開く ---
fileopen fhandle INI_FILE 0
if fhandle == -1 then
    messagebox "ファイルが見つかりません" "Error"
    end
endif

:loop
; --- 1行読み込み ---
filereadln fhandle line
if result == 1 goto fclose

; --- '=' で分割する ---
strscan line '='
if result > 0 then
    ; 1文字目から '=' の直前までを「変数名」として取得
    strcopy line 1 result-1 var_name
    ; '=' の直後から最後までを「値」として取得
    strcopy line result+1 512 var_value
    
    ; --- 取得した値を変数にセットするロジック ---
    ; TTLでは動的な変数名定義が難しいため、if文で振り分けるのが確実です
    strcompare var_name 'HOST_IP'
    if result == 0 HOST_IP = var_value

    strcompare var_name 'USER_NAME'
    if result == 0 USER_NAME = var_value

    strcompare var_name 'PASS_WORD'
    if result == 0 PASS_WORD = var_value
endif

goto loop

:fclose
fileclose fhandle

; --- 確認用 ---
messagebox HOST_IP "読み込んだIPアドレス"
