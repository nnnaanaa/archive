; ============================================================
; 1. 環境設定と変数宣言
; ============================================================
CMD_FILE    = 'P:\teraterm\commands.txt'
LOG_DIR     = 'P:\teraterm\'

JUMP_HOST   = '10.0.0.25'
JUMP_USER   = 'root'
JUMP_PASS   = 'nanahira'

TARGET_IP   = '10.0.0.21'
TARGET_USER = 'nanahira'
TARGET_PASS = 'nanahira'

; --- 宣言と初期化 (Error 9/40 対策) ---
strdim COMMAND_LIST 21
SELECTED_CMD = ''
STR_CMD = ''
TIMESTAMP = ''
LOG_FILE = ''

; ============================================================
; 2. 接続処理
; ============================================================
COMMAND = JUMP_HOST
strconcat COMMAND ':22 /ssh /2 /auth=password /user='
strconcat COMMAND JUMP_USER
strconcat COMMAND ' /passwd='
strconcat COMMAND JUMP_PASS

connect COMMAND
if result <> 2 then
    messagebox '接続に失敗しました。' 'Error'
    end
endif

timeout = 10
wait '#' '$' '>'

sendln 'ssh ' TARGET_USER '@' TARGET_IP
wait 'password:'
sendln TARGET_PASS
wait '#' '$' '>'

; ============================================================
; 3. メインループの起点 (Error 205 対策)
; ============================================================
:main_menu

fileopen fhandle CMD_FILE 0
if fhandle == -1 then
    messagebox '定義ファイルが見つかりません。' 'Error'
    goto do_exit
endif

count = 0
:loop_file
filereadln fhandle line
if result then
    goto close_file
endif
strlen line
if result > 0 then
    COMMAND_LIST[count] = line
    count = count + 1
endif
if count < 20 goto loop_file

:close_file
fileclose fhandle
COMMAND_LIST[count] = '[EXIT] マクロを終了'

listbox '実行するコマンドを選択してください' 'Command Menu' COMMAND_LIST
if result == -1 goto do_exit
if result == count goto do_exit

; --- ここで変数を確実にセットする ---
SELECTED_CMD = COMMAND_LIST[result]
STR_CMD = SELECTED_CMD

; ============================================================
; 4. ログ記録とコマンド実行
; ============================================================
gettime TIMESTAMP "%Y%m%d%H%M%S"

; --- 記号置換処理 (スラッシュが複数あってもOKなループ) ---
:replace_slash
strreplace STR_CMD 1 "/" "_"
if result != 0 goto replace_slash

:replace_space
strreplace STR_CMD 1 " " "_"
if result != 0 goto replace_space

LOG_FILE = LOG_DIR
strconcat LOG_FILE TIMESTAMP
strconcat LOG_FILE "_"
strconcat LOG_FILE STR_CMD
strconcat LOG_FILE ".log"

logclose
flushrecv
pause 1

; ログ開始
logopen LOG_FILE 0 0
mpause 500

; ★1行目にプロンプトを入れるための「空改行」
sendln ""
wait '#' '$' '>'

; 本命のコマンド実行
sendln SELECTED_CMD

; 出力完了まで待機
timeout = 30
wait '#' '$' '>'

; 最後のプロンプトを書き込ませるための猶予
mpause 500
logclose
pause 1

goto main_menu

; ============================================================
; 5. 終了処理
; ============================================================
:do_exit
logclose
sendln 'exit'
wait '#' '$' '>'
sendln 'exit'
closett
end
