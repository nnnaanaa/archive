TITLE = 'ホスト接続・設定確認マクロ'
strdim CATEGORY_LIST 10
strdim TYPE_LIST 10
CONFIG_FILE = 'item_list.txt'
LOG_FILE = 'C:\temp\macro_log.txt'

; 設定ファイルの存在確認
filesearch CONFIG_FILE

; 接続情報の設定
HOST_IP   = '10.0.0.25'
USERNAME  = 'root'
PASSWORD  = 'nanahira'
sprintf2 conn_cmd '%s:22 /ssh /2 /auth=password /user=%s /passwd=%s' HOST_IP USERNAME PASSWORD
connect conn_cmd

; 接続失敗時の処理
if result <> 2 then
    messagebox 'ホストへの接続に失敗しました。' 'Error'
    end
endif

; プロンプトを待機してログ取得開始
wait '#' '$' '>'
logopen LOG_FILE 0 0
mpause 500

; コマンド実行（プロンプト確保のための空改行含む）
sendln ""
wait '#' '$' '>'

sendln 'hostname'
wait '#' '$' '>'
logclose

; ファイル検索結果（filesearch）の確認
if result = 0 then
    messagebox '設定ファイルが見つかりません。' '確認'
    goto end_label
endif

:loop_main
; 選択肢の定義
CATEGORY_LIST[0] = '拠点A：拠点管理サーバー'
CATEGORY_LIST[1] = '拠点B：開発用サーバー'
CATEGORY_LIST[2] = '拠点C：バックアップサーバー'
CATEGORY_LIST[3] = '拠点D：外部接続用サーバー'

listbox '管理対象の拠点を選択してください' TITLE CATEGORY_LIST
if result == -1 goto end_label

SELECTED_INDEX = result
TARGET_NAME = CATEGORY_LIST[SELECTED_INDEX]

; 選択結果の表示
msg = TARGET_NAME
strconcat msg ' が選択されました。設定を確認しますか？'
yesnobox msg TITLE

if result = 0 goto loop_main

; 設定ファイル（item_list.txt）の読み込みと表示
fileopen fhandle CONFIG_FILE 0

:read_loop
filereadln fhandle line
if result = 1 goto close_file
messagebox line '設定項目'
goto read_loop

:close_file
fileclose fhandle

; 再度メインループに戻るか終了するか
yesnobox '他の拠点も確認しますか？' TITLE
if result = 1 goto loop_main

:end_label
end
